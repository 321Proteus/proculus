# Signature matching for web services. Regular expressions run under PCRE2's
# match limits, so hostile pages cannot make a pattern run away.
find_package(PkgConfig REQUIRED)
pkg_check_modules(PCRE2 REQUIRED IMPORTED_TARGET libpcre2-8)

# The built-in signatures (builtin/*.json) are compiled into the library.
file(GLOB BUILTIN_JSON CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/builtin/*.json)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${BUILTIN_JSON})
set(BUILTIN_CPP ${CMAKE_CURRENT_BINARY_DIR}/builtin_data.cpp)
set(_gen "// Generated by CMake from libs/signatures/builtin/*.json. Do not edit.\n#include \"proculus/sig/builtin.hpp\"\n\nnamespace proculus::sig {\n\nconst std::vector<BuiltinSource>& builtin_sources() {\n\tstatic const std::vector<BuiltinSource> sources = {\n")
foreach(f ${BUILTIN_JSON})
	get_filename_component(_name ${f} NAME)
	file(READ ${f} _content)
	string(APPEND _gen "\t\t{\"${_name}\", R\"PROCSIG(${_content})PROCSIG\"},\n")
endforeach()
string(APPEND _gen "\t};\n\treturn sources;\n}\n\n} // namespace proculus::sig\n")
file(WRITE ${BUILTIN_CPP} "${_gen}")

add_library(proculus_sig STATIC
	src/regex.cpp
	src/evidence.cpp
	src/fields.cpp
	src/evidence_json.cpp
	src/loader.cpp
	src/engine.cpp
	src/builtin.cpp
	${BUILTIN_CPP})
target_include_directories(proculus_sig PUBLIC include)
target_link_libraries(proculus_sig PUBLIC nlohmann_json::nlohmann_json PRIVATE PkgConfig::PCRE2)

if(BUILD_TESTING)
	add_executable(test_sig_regex tests/test_regex.cpp)
	target_link_libraries(test_sig_regex PRIVATE proculus_sig)
	add_test(NAME sig_regex COMMAND test_sig_regex)

	add_executable(test_signatures tests/test_signatures.cpp)
	target_link_libraries(test_signatures PRIVATE proculus_sig)
	add_test(NAME sig_engine COMMAND test_signatures)

	add_executable(test_builtin tests/test_builtin.cpp)
	target_link_libraries(test_builtin PRIVATE proculus_sig)
	add_test(NAME sig_builtin COMMAND test_builtin)
endif()
