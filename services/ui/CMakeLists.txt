# The pages (web/*) are compiled into the binary, like the built-in signatures.
file(GLOB UI_FILES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/web/*)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${UI_FILES})
set(UI_CPP ${CMAKE_CURRENT_BINARY_DIR}/ui_data.cpp)
# Embedded as byte arrays, not string literals: a page can be bigger than the 64 KiB a string literal is guaranteed to hold.
set(_decl "")
set(_table "")
set(_n 0)
foreach(f ${UI_FILES})
	get_filename_component(_name ${f} NAME)
	get_filename_component(_ext ${f} EXT)
	if(_ext STREQUAL ".html")
		set(_mime "text/html; charset=utf-8")
	elseif(_ext STREQUAL ".js")
		set(_mime "text/javascript; charset=utf-8")
	elseif(_ext STREQUAL ".css")
		set(_mime "text/css; charset=utf-8")
	elseif(_ext STREQUAL ".svg")
		set(_mime "image/svg+xml")
	else()
		message(FATAL_ERROR "ui: no MIME type for ${_name}")
	endif()
	file(READ ${f} _hex HEX)
	string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _bytes "${_hex}")
	string(APPEND _decl "static const unsigned char asset_${_n}[] = {${_bytes}0x00};\n")
	string(APPEND _table "\t\t{\"/${_name}\", \"${_mime}\", reinterpret_cast<const char*>(asset_${_n})},\n")
	math(EXPR _n "${_n}+1")
endforeach()
set(_gen "// Generated by CMake from services/ui/web/*. Do not edit.\n#include \"ui_assets.hpp\"\n\nnamespace proculus {\n\nnamespace {\n${_decl}} // namespace\n\nconst std::vector<UiAsset>& ui_assets() {\n\tstatic const std::vector<UiAsset> assets = {\n${_table}\t};\n\treturn assets;\n}\n\n} // namespace proculus\n")
file(WRITE ${UI_CPP} "${_gen}")

find_package(Threads REQUIRED)
add_library(ui_core STATIC src/ui_server.cpp ${UI_CPP})
target_include_directories(ui_core PUBLIC src)
target_include_directories(ui_core SYSTEM PUBLIC ${PROJECT_SOURCE_DIR}/third_party/httplib)
target_link_libraries(ui_core PUBLIC proculus_web Threads::Threads)

add_executable(ui src/main.cpp)
target_link_libraries(ui PRIVATE ui_core)

if(BUILD_TESTING)
	add_executable(test_ui tests/test_ui.cpp)
	target_link_libraries(test_ui PRIVATE ui_core)
	add_test(NAME ui_server COMMAND test_ui)
endif()
